# Oh My Zsh

1. Clone this repository into $ZSH_CUSTOM/plugins (by default ~/.oh-my-zsh/custom/plugins)
    ```
    git clone https://github.com/zsh-users/zsh-autosuggestions ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions
    ```

2. Add the plugin to the list of plugins for Oh My Zsh to load (inside ~/.zshrc):

    ```
        export ZSH="$HOME/.oh-my-zsh"

        ZSH_AUTOSUGGEST_STRATEGY=(history completion)

        plugins=(
          git
          zsh-autosuggestions
        )

        source $ZSH/oh-my-zsh.sh

        PROMPT='%F{green}➜%f  %F{cyan}%~%f $(git_prompt_info)'
    ```

3. Start a new terminal session.

# Homebrew

1. Install command:
    ```
        brew install zsh-autosuggestions
    ```

2. To activate the autosuggestions, add the following at the end of your .zshrc:
    ```
        source $(brew --prefix)/share/zsh-autosuggestions/zsh-autosuggestions.zsh
    ```

3. Start a new terminal session.

# Termux

```bash
cat > setup-termux-zsh.sh <<'EOF'
#!/usr/bin/env bash
set -euo pipefail

echo "======================================"
echo " Termux zsh setup script"
echo "======================================"

echo
echo "[1/7] Update packages..."
pkg update -y
pkg upgrade -y

echo
echo "[2/7] Install zsh and git..."
pkg install -y zsh git

ZSH_PATH="$(command -v zsh)"
echo "zsh path: $ZSH_PATH"

echo
echo "[3/7] Prepare plugin directory..."
mkdir -p "$HOME/.zsh/plugins"

clone_or_update() {
  local repo_url="$1"
  local target_dir="$2"

  if [ -d "$target_dir/.git" ]; then
    echo "Plugin already exists: $target_dir"
    echo "Updating..."
    git -C "$target_dir" pull --ff-only || true
  else
    echo "Cloning $repo_url"
    rm -rf "$target_dir"
    git clone "$repo_url" "$target_dir"
  fi
}

echo
echo "[4/7] Install zsh-autosuggestions..."
clone_or_update \
  "https://github.com/zsh-users/zsh-autosuggestions" \
  "$HOME/.zsh/plugins/zsh-autosuggestions"

echo
echo "[5/7] Install zsh-completions..."
clone_or_update \
  "https://github.com/zsh-users/zsh-completions" \
  "$HOME/.zsh/plugins/zsh-completions"

echo
echo "[6/7] Backup old ~/.zshrc and create new one..."

if [ -f "$HOME/.zshrc" ]; then
  BACKUP_FILE="$HOME/.zshrc.bak.$(date +%Y%m%d%H%M%S)"
  cp "$HOME/.zshrc" "$BACKUP_FILE"
  echo "Backup created: $BACKUP_FILE"
fi

cat > "$HOME/.zshrc" <<'ZSHRC'
# ==================================================
# Termux zsh config
# Generated by setup-termux-zsh.sh
# ==================================================

# ---------- history ----------
HISTFILE="$HOME/.zsh_history"
HISTSIZE=10000
SAVEHIST=10000

setopt appendhistory
setopt sharehistory
setopt hist_ignore_dups
setopt hist_ignore_space

# ---------- completion ----------
fpath=("$HOME/.zsh/plugins/zsh-completions/src" $fpath)

autoload -Uz compinit
compinit -u -d "$HOME/.zcompdump"

zmodload zsh/complist 2>/dev/null || true

zstyle ':completion:*' menu select
zstyle ':completion:*' matcher-list 'm:{a-zA-Z}={A-Za-z}'
zstyle ':completion:*' list-dirs-first true

# ---------- autosuggestions ----------
ZSH_AUTOSUGGEST_STRATEGY=(history completion)
ZSH_AUTOSUGGEST_HIGHLIGHT_STYLE='fg=244'

if [ -f "$HOME/.zsh/plugins/zsh-autosuggestions/zsh-autosuggestions.zsh" ]; then
  source "$HOME/.zsh/plugins/zsh-autosuggestions/zsh-autosuggestions.zsh"
fi

# ---------- git branch prompt ----------
autoload -Uz vcs_info

precmd() {
  vcs_info
}

zstyle ':vcs_info:git:*' formats ' (%b)'

setopt PROMPT_SUBST

# ---------- prompt ----------
PROMPT='%F{green}➜%f  %F{cyan}%~%f%F{yellow}${vcs_info_msg_0_}%f %# '

# ---------- useful aliases ----------
alias ll='ls -la'
alias la='ls -A'
alias l='ls -CF'
alias cls='clear'
ZSHRC

echo
echo "[7/7] Set zsh as default shell..."

if command -v chsh >/dev/null 2>&1; then
  chsh -s "$ZSH_PATH" || {
    echo "chsh failed. You can run this manually:"
    echo "chsh -s $ZSH_PATH"
  }
else
  echo "chsh command not found."
  echo "You can still start zsh manually with:"
  echo "exec zsh"
fi

echo
echo "======================================"
echo " Setup completed"
echo "======================================"
echo
echo "Run this command now:"
echo
echo "  exec zsh"
echo
echo "Then test autosuggestions:"
echo
echo "  echo hello"
echo "  ec"
echo
echo "If gray suggestion appears, it works."
EOF
```
